sget_period
	; IN	d0=octave (0-8)
	;	d1=note
	;	d2=ln2(wavesize)
	; OUT	d3=period
	; uses	d0/d1
	subq.w	#2,d0
	add.w	d2,d0
	add.w	d1,d1
	move.w	note_table(pc,d1.w),d3
	lsr.w	d0,d3
	rts

	; shift right by octave+ln2(wavesize)-2
	; 0123,4,5678 is middle octave
note_table
	dc.w	$d5c7,$c9c8,$be74	c  c# d
	dc.w	$b3c4,$a9ad,$a027	d# e  f
	dc.w	$972a,$8eae,$86ac	f# g  g#
	dc.w	$7f1d,$77fa,$713f	a  a# b
